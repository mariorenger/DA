\section{Vấn đề}
%% ============================================================================ Vấn đề
\begin{frame}[shrink]{Vấn đề: lời giải nháp có thật sự được dùng để ra đáp án?}
\begin{columns}[T,onlytextwidth]
\column{0.50\textwidth}
\footnotesize
\textbf{Hình dung một học sinh làm toán.} Em viết các bước nháp rồi ghi đáp án. Có hai khả năng:
\begin{enumerate}
  \setlength{\itemsep}{3pt}
  \item Em \emph{tính} đáp án từ các bước nháp (bỏ một bước nháp thì đáp án sai).
  \item Em đã biết hoặc đoán đáp án từ đầu, nháp chỉ để cho đẹp (bỏ nháp đi thì đáp án vẫn vậy).
\end{enumerate}
\vspace{3pt}
Mô hình ngôn ngữ cũng ``viết nháp từng bước'' trước khi trả lời. Kỹ thuật này gọi là \textbf{chain-of-thought (CoT)}. Đề tài hỏi: \textbf{mô hình thuộc khả năng 1 hay 2?}
\column{0.44\textwidth}
\begin{block}{Vì sao câu hỏi này quan trọng}
\footnotesize
\begin{itemize}
  \setlength{\itemsep}{3pt}
  \item Nếu mô hình không dựa vào nháp, thì kiểm tra hay sửa nháp không giúp được gì.
  \item Muốn làm mô hình trả lời đúng hơn, ta phải biết can thiệp vào \emph{chỗ nào}: lúc viết nháp, hay lúc chốt đáp án.
  \item Mô hình nhỏ (0.5--7 tỷ tham số) rẻ, dễ triển khai, nhưng chưa rõ chúng suy luận ra sao.
\end{itemize}
\end{block}
\end{columns}
\par\kern0pt
\note{Dùng phép ví dụ học sinh làm toán. Khả năng 1 là suy luận thật, khả năng 2 là viết nháp trang trí. Toàn bộ nghiên cứu là cách phân biệt hai khả năng này một cách có kiểm chứng.}
\end{frame}

%% ============================================================================ hai loại câu hỏi
\begin{frame}[shrink]{Hai loại bài toán: một loại ``chép được'', một loại ``phải tính''}
\begin{columns}[T,onlytextwidth]
\column{0.46\textwidth}
\begin{block}{Loại A: đáp án nằm sẵn trong nháp (MuSiQue)}
\scriptsize
\textbf{Hỏi:} Vợ của đạo diễn phim X là ai?\par\vspace{2pt}
\code{Bước 1: Đạo diễn phim X là Y.}\\
\code{Bước 2: Vợ của Y là Z.}\\
\code{Đáp án: Z}\par\vspace{2pt}
Mỗi bước tìm ra một thông tin, bước sau dùng nó. Chữ ``Z'' được \textbf{viết ra} ở bước cuối: chỉ cần \emph{đọc lại} bước cuối là có đáp án.
\end{block}
\column{0.50\textwidth}
\begin{block}{Loại B: đáp án phải tự tính (CLUTRR, quan hệ họ hàng)}
\scriptsize
\textbf{Chuyện:} Anna đưa con trai Ben đi chơi. Ben và chị gái Clara nướng bánh. \textbf{Hỏi:} Clara là gì của Anna?\par\vspace{2pt}
\code{Bước 1: Ben là con trai của Anna.}\\
\code{Bước 2: Clara là chị gái của Ben.}\\
\code{Đáp án: con gái}\par\vspace{2pt}
Chữ ``con gái'' \textbf{không có} trong các bước: phải tự \emph{tính} ``con trai + chị gái = con gái''.
\end{block}
\end{columns}
\vspace{0.2cm}
\begin{alertblock}{Dùng hai loại này để phân biệt}
\footnotesize
Nếu mô hình chỉ dựa vào nháp khi đáp án \emph{chép được} (loại A), và không dựa vào nháp khi đáp án \emph{phải tính} (loại B), thì nó đang \textbf{đọc lại} chứ không \textbf{suy luận}.
\end{alertblock}
\par\kern0pt
\note{Hai loại bài toán trông giống nhau nhưng khác ở chỗ đáp án có xuất hiện trong lời giải nháp hay không. Đó là chìa khoá của thiết kế.}
\end{frame}

\begin{frame}[shrink]{Cách đo: ``rút một viên gạch'' rồi xem tòa nhà có đổ không}

\end{tikzpicture}
\par\vspace{3pt}
\begin{columns}[T,onlytextwidth]
\column{0.46\textwidth}
\begin{block}{Hai khái niệm}
\scriptsize
\textbf{Độ tin vào đáp án đúng} ($p$): xác suất (0--100\%) mô hình gán cho đáp án đúng.\par
\textbf{Hiệu ứng} $\Delta$ = $p$(nháp đầy đủ) $-$ $p$(nháp đã sửa), tính bằng \emph{điểm phần trăm}. Ở ví dụ trên $\Delta = 90-20 = 70$ điểm.
\end{block}
\column{0.50\textwidth}
\begin{exampleblock}{Cách đọc $\Delta$}
\scriptsize
$\Delta$ \textbf{lớn}: đáp án \emph{phụ thuộc} vào bước bị che.\par
$\Delta \approx 0$: bỏ bước đó đi, mô hình vẫn trả lời như cũ, tức là \emph{không dùng} bước đó.
\end{exampleblock}
\end{columns}
\par\kern0pt
\note{Phương pháp là can thiệp: sửa một chỗ trong nháp rồi đo độ tin vào đáp án đúng thay đổi bao nhiêu. Delta lớn nghĩa là đáp án phụ thuộc vào chỗ đó.}
\end{frame}

%% ============================================================================ ground truth
\begin{frame}[shrink]{Làm sao biết phép đo đúng? Dùng đề có đáp án biết trước}
\begin{columns}[T,onlytextwidth]
\column{0.50\textwidth}
\footnotesize
\textbf{Vấn đề:} nhiều nghiên cứu đo ``bước nào quan trọng'' bằng cách che từng bước. Nhưng \emph{không ai biết bước nào thật sự quan trọng}, nên không kiểm chứng được phép đo.\par\vspace{4pt}
\textbf{Cách giải quyết của đề tài:} tự tạo tình huống mà ta \emph{biết chắc} kết quả đúng.
\vspace{3pt}
\renewcommand{\arraystretch}{1.2}
\scriptsize
\begin{tabularx}{\linewidth}{@{}P{3.2cm}Y@{}}
\toprule
\textbf{Tình huống} & \textbf{Phép đo tốt phải báo}\\
\midrule
Chèn một câu \emph{đúng nhưng vô quan hệ} & $\approx 0$ (câu này thừa)\\
Bước gold thật sự cần thiết & $\Delta$ lớn\\
Đổi tên nhân vật (Anna $\to$ Lan) & $\approx 0$ (đáp án vẫn vậy)\\
\bottomrule
\end{tabularx}
\column{0.44\textwidth}
\begin{block}{Ví dụ: chèn một bước thừa}
\scriptsize
\code{Bước 1: Ben là con trai của Anna.}\\
\code{Bước 2: Clara là chị gái của Ben.}\\
\textcolor{failc}{\code{Bước 3: Anna thích bóng đá.}} \emph{(đúng, nhưng không liên quan)}\par\vspace{2pt}
Mức cần thiết thật của bước 3 là \textbf{0}. Nếu một phép đo chấm bước 3 là ``quan trọng'' thì phép đo đó \textbf{sai}.
\end{block}
\begin{exampleblock}{AUC: một điểm số cho chất lượng phép đo}
\scriptsize
Xếp các bước theo điểm quan trọng. AUC $=$ xác suất bước cần thiết được xếp cao hơn bước thừa. $1.0$ là hoàn hảo, $0.5$ là đoán mò.
\end{exampleblock}
\end{columns}
\par\kern0pt
\note{Vì ta tự chèn bước thừa, ta biết chắc mức cần thiết thật của nó bằng không. Phép đo nào chấm bước thừa là quan trọng thì phép đo đó sai. AUC tóm tắt điều này bằng một con số.}
\end{frame}

Tín hiệu học thưa: chỉ khoảng 1 trong 8 nhóm có lời giải khác điểm nhau, vì mô hình đã giải đúng $\sim$90\% bài ngắn. \FAILs
\end{exampleblock}
\par\kern0pt
\note{GRPO là kỹ thuật học tăng cường đang phổ biến. Ở đây phần thưởng kiểm chứng được. Kết quả: không giúp chuỗi dài, nhưng làm đáp án bền hơn trước nhiễu. Một lần chạy, nên kết luận còn thận trọng.}

%% ============================================================================ Kết luận
\begin{frame}[shrink]{Kết luận: chứng minh được gì, chưa chứng minh được gì}
\begin{columns}[T,onlytextwidth]
\column{0.495\textwidth}
\begin{block}{Chứng minh được}
\scriptsize
\begin{enumerate}
  \setlength{\itemsep}{3pt}
  \item Hai cách đo ``step importance'' phổ biến (xoá một bước; resampling baseline cố định) \textbf{lệch có hệ thống}, có công thức đóng và được xác nhận (R1, R2).
  \item Mô hình 0.5B--7B \textbf{đọc} chứ không \textbf{suy luận qua} chuỗi; $\le$3B chép cả đáp án sai mà nó tự giải được (R3).
  \item Tác hại của distractor đi qua chuỗi được sinh lại (R4), nên decoding ở bước đọc không sửa được (R5).
  \item Thực thi phép hợp thành ngoài mô hình tăng accuracy rất lớn, và ablation chứng minh bước hợp thành là nguyên nhân (D2, D3).
  \item \textbf{Fine-tune thường} trên chuỗi gold dạy được phép hợp thành (CLUTRR 3B: 36\% $\to$ 95\%, hơn D2) và chuyển sang văn bản thật (2Wiki: $+21$ đến $+30$ so với few-shot); thêm cấu trúc hay trích dẫn nguồn không thêm gì (D4, D5).
  \item Sau fine-tune mô hình \textbf{thật sự dùng các bước} (hiệu ứng che bước 1--7 $\to$ $\sim$40 điểm) và viết chuỗi hơn chỉ-đáp-án 9--23 điểm ở đề lạ hoặc dài (D6, D7).
\end{enumerate}
\end{block}
\column{0.465\textwidth}
\begin{block}{Chưa chứng minh được}
\scriptsize
\begin{itemize}
  \setlength{\itemsep}{3pt}
  \item Một phương pháp \emph{decoding token} mới vượt self-consistency: HGCD, CAD, calibration, lọc bằng attention đều không đạt.
  \item Extract + rules mới đúng ở miền có luật đóng (CLUTRR).
  \item Cách vừa chống chép đáp án sai vừa giữ độ chính xác chuỗi dài (DPO chưa đạt, GRPO chưa giúp).
  \item Tổng quát ngoài phân phối và sang MuSiQue (chưa chạy); chỉ mới kiểm tra 2Wiki cùng phân phối.
  \item Cấu trúc chuỗi có ích (tiêu chí chính không đạt).
\end{itemize}
\end{block}
\end{columns}
\par\vspace{3pt}
\begin{alertblock}{Thông điệp}
\footnotesize Mô hình nhỏ chưa huấn luyện \emph{đọc} chuỗi chứ không \emph{suy luận qua} chuỗi, nên chỉnh cách chọn token không giúp. Cái giúp là \textbf{can thiệp vào chỗ bị hỏng}: đưa phép hợp thành ra ngoài, hoặc dạy mô hình bằng chuỗi có nhãn. Kết quả cuối cùng thực dụng nhất là fine-tune thường; phần nhân quả cần chứng minh tiếp là mô hình sau fine-tune có dùng chuỗi hay không.
\end{alertblock}
\par\kern0pt
\note{Em tóm tắt lại bốn điều đã chứng minh và bốn điều chưa. Thông điệp: tìm đúng chỗ hỏng rồi mới can thiệp.}
\end{frame}

\section{Phụ lục}
%% ============================================================================ Thuật ngữ
\begin{frame}[shrink]{Phụ lục \textperiodcentered{} Thuật ngữ}
\begin{columns}[T,onlytextwidth]
\column{0.461\textwidth}
\scriptsize
\begin{description}
  \setlength{\itemsep}{1pt}
  \item[CoT] chain-of-thought: chuỗi suy luận từng bước mô hình viết ra trước khi trả lời.
  \item[Hop] một bước trong chuỗi, dạng \code{Hop i: nội dung $\Rightarrow$ kết quả}.
  \item[DRG] đồ thị suy luận có hướng: nối bước $i\to j$ nếu kết quả của $i$ được nhắc lại trong $j$.
  \item[$\dn$, $\dt$] che một bước / che bước cùng các bước phụ thuộc.
  \item[Placebo, marker] control: viết lại cùng nghĩa / chèn \code{[MASKED]} nhưng giữ nội dung.
  \item[AUC] khả năng phân biệt bước cần thiết với bước thừa (0.5 là đoán mò).
  \item[PS $=1$] câu hỏi mà mô hình tự giải đúng từ chuỗi gold.
\end{description}
\column{0.451\textwidth}
\scriptsize
\begin{description}
  \setlength{\itemsep}{1pt}
  \item[TE / NDE / TIE] hiệu ứng tổng / trực tiếp (chuỗi cố định) / gián tiếp (qua chuỗi).
  \item[SC] self-consistency: sinh nhiều chuỗi, bỏ phiếu đa số; ``cùng compute'' nghĩa là baseline có ngân sách sinh không ít hơn.
  \item[HGCD, CAD] decoding ở bước đọc đáp án: khuếch đại hiệu ứng của chuỗi / tương phản có và không có ngữ cảnh.
  \item[Calibration] trừ phần thiên lệch của mô hình khi không có nội dung.
  \item[Go/no-go] ngưỡng thành công ghi trước khi chạy.
  \item[LoRA] fine-tune nhẹ bằng các ma trận hạng thấp.
  \item[Rules-in-prompt] đưa bảng luật vào prompt thay vì thực thi ngoài mô hình.
\end{description}
\end{columns}
