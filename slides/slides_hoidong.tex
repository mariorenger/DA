\section{Bối cảnh và mục tiêu}
%% ============================================================================ 3
\begin{frame}{Bối cảnh và vấn đề}
\textbf{Bối cảnh}
\begin{itemize}
  \item Chain-of-Thought (CoT): mô hình viết các bước trung gian $Z = (z_1, \dots, z_k)$ trước khi đưa ra đáp án $Y$.
  \item Một chuỗi đúng không bảo đảm đáp án được tính từ chuỗi đó. Mô hình có thể trả lời theo lối tắt và viết chuỗi như phần diễn giải đi kèm (Lanham et al., 2023; Paul et al., 2024).
  \item Mô hình nhỏ (0.5B đến 8B) được triển khai rộng rãi vì chi phí thấp và chạy được tại chỗ, nhưng ít được phân tích về mặt này.
\end{itemize}
\vspace{4pt}
\textbf{Vấn đề nghiên cứu}
\begin{itemize}
  \item Các phép đo hiện có (che một bước, lấy mẫu lại) chưa được kiểm chứng trên dữ liệu có kết quả biết trước.
  \item Chưa tách được hai năng lực: đọc thông tin có sẵn và tự hợp thành thông tin mới.
  \item Chưa rõ nên can thiệp ở giai đoạn giải mã hay giai đoạn huấn luyện.
\end{itemize}
\end{frame}

%% ============================================================================ 4
\begin{frame}{Mục tiêu và đóng góp}
\textbf{Mục tiêu}
\begin{itemize}
  \item Xây dựng phép đo mức độ mô hình dùng chuỗi suy luận, kiểm chứng bằng các can thiệp có kết quả biết trước.
  \item Dùng phép đo đó để chẩn đoán mô hình nhỏ, sau đó tìm vị trí can thiệp hiệu quả.
\end{itemize}
\vspace{4pt}
\textbf{Đóng góp}
\begin{enumerate}
  \item \textbf{Phương pháp đo.} Chỉ ra hai phép đo phổ biến cho kết quả lệch và cách hiệu chỉnh.
  \item \textbf{Chẩn đoán.} Mô hình nhỏ đọc đáp án có sẵn trong chuỗi nhưng không dùng chuỗi để hợp thành quan hệ.
  \item \textbf{Can thiệp.} Can thiệp ở bước giải mã cuối không hiệu quả. Hai cách hiệu quả là giao phép hợp thành cho bộ luật và tinh chỉnh trên chuỗi.
  \item \textbf{Ứng dụng.} Dữ liệu huấn luyện có tài liệu thiếu bằng chứng và tài liệu mâu thuẫn giúp mô hình biết từ chối và bám theo tài liệu.
\end{enumerate}
\end{frame}

\begin{itemize}
  \item Hiệu ứng của bước $z_i$ là mức giảm xác suất mô hình gán cho đáp án đúng $Y^*$ khi bước đó bị can thiệp:
  $$\Delta_i = p(Y^* \mid X, Z) - p\big(Y^* \mid X, \mathrm{do}(z_i)\big)$$
  \item $\Delta_i$ lớn: mô hình dựa vào bước $z_i$ khi trả lời. $\Delta_i \approx 0$: mô hình bỏ qua bước đó.
  \item Kiểm chứng phép đo: chèn vào chuỗi một bước đúng nhưng không liên quan tới câu hỏi. Hiệu ứng thật của bước này bằng 0, nên phép đo đúng phải cho $\Delta \approx 0$.
\end{itemize}
\end{frame}

\section{Giai đoạn 1. Chẩn đoán}
%% ============================================================================ 9
\begin{frame}[shrink]{TN1. Phép che một bước không phát hiện được bước bắc cầu}
\small
\textbf{Giả thuyết.} Che riêng một bước giữa (Lanham et al., 2023) cho $\Delta \approx 0$ vì bước cuối vẫn ghi sẵn đáp án.\par
\textbf{Thiết kế.} Trên MuSiQue so sánh: che riêng một bước ($\dn$), che bước và mọi bước phụ thuộc vào nó ($\dt$), và bỏ toàn bộ chuỗi. Ngữ cảnh $X$ luôn giữ nguyên.\par
\vspace{4pt}
\footnotesize
\renewcommand{\arraystretch}{1.2}
\begin{tabularx}{\textwidth}{@{}YQ{1.9cm}Q{1.9cm}Q{1.9cm}Q{2.3cm}@{}}
\toprule
\textbf{Mô hình} & $\Delta$ của $\dn$ & $\Delta$ của $\dt$ & Bỏ cả chuỗi & AUROC của $\dn$\\
\midrule
Qwen2.5-0.5B & $-2.4$ & 56.7 & 52.6 & 0.57\\
Qwen2.5-1.5B & 0.6 & 56.9 & 55.2 & 0.71\\
SmolLM2-1.7B & $-3.9$ & 66.8 & 61.0 & 0.46\\
Qwen2.5-3B & 0.5 & 49.1 & 50.6 & 0.64\\
Qwen2.5-7B & 6.5 & 41.3 & 37.7 & 0.63\\
\bottomrule
\end{tabularx}
\nguon{$\Delta$ tính bằng điểm phần trăm trên các bước giữa. AUROC phân biệt bước giữa với bước không liên quan, 0.5 là ngẫu nhiên.}
\vspace{2pt}
\small
\textbf{Kết quả.}
\begin{itemize}
  \item $\dn$ cho $\Delta \approx 0$ và AUROC gần mức ngẫu nhiên.
  \item $\dt$ xấp xỉ hiệu ứng bỏ cả chuỗi, tức là đo việc đáp án có trong chuỗi, không đo riêng vai trò một bước.
\end{itemize}
\ketluan{Phép che một bước bị mù với bước bắc cầu. Dù ngữ cảnh có sẵn đáp án, mô hình lấy đáp án từ chuỗi suy luận.}
\end{frame}

  \item Mô hình vẫn tin dòng kết luận cuối. Với 7B, tỷ lệ chép kết luận sai tăng sau tinh chỉnh.
  \item Huấn luyện thêm bằng DPO với cặp phản thực giảm tỷ lệ chép xuống 4.5\% đến 7.2\%, nhưng làm giảm 5 đến 13 điểm trên chuỗi dài. GRPO không cải thiện chuỗi dài.
\end{itemize}
\ketluan{Tinh chỉnh thay đổi cơ chế: mô hình chuyển sang dựa vào chuỗi. DPO và GRPO chỉ cho đánh đổi, không cho cải thiện thêm.}
\end{frame}
