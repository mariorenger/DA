\section{Đặt vấn đề}
%% ============================================================================ Vấn đề
\begin{frame}[shrink]{Đặt vấn đề: tính trung thực nhân quả của chuỗi suy luận}
\footnotesize
\begin{columns}[T,onlytextwidth]
\column{0.52\textwidth}
\textbf{\color{keyc}Bối cảnh nghiên cứu}\par\vspace{3pt}
Kỹ thuật \textbf{Chain-of-Thought (CoT)} cải thiện hiệu năng suy luận bằng cách sinh chuỗi bước trung gian $Z = (z_1, \dots, z_k)$ từ ngữ cảnh $X$ trước khi đưa ra đáp án $Y$. Nhiều nghiên cứu gần đây đặt vấn đề về \textbf{tính trung thực (faithfulness)} của $Z$ \citeref{Lanham et al., 2023; Turpin et al., NeurIPS 2023; Paul et al., Findings of EMNLP 2024}. Có hai cơ chế đối lập:
\begin{enumerate}
  \setlength{\itemsep}{4pt}
  \item \textbf{Suy diễn nhân quả ($X \to Z \to Y$):} đáp án $Y$ là kết quả tính toán từ các bước $z_i$; thay đổi $z_i$ kéo theo thay đổi $Y$.
  \item \textbf{Hợp thức hóa hậu nghiệm ($X \to Y$):} mô hình dự đoán dựa trên lối tắt thống kê hoặc chỉ sao chép thực thể ở bước cuối, còn $Z$ chỉ là phần diễn giải đi kèm.
\end{enumerate}

\column{0.44\textwidth}
\textbf{\color{keyc}Khoảng trống nghiên cứu}\par\vspace{3pt}
\begin{itemize}
  \setlength{\itemsep}{4pt}
  \item \textbf{Thiếu nhãn chuẩn.} Các công trình trước đo tầm quan trọng của $z_i$ trên chuỗi tự sinh nhưng không biết giá trị thực để kiểm chuẩn thước đo. Hướng dùng tác vụ có hàm xác định, trong đó mỗi can thiệp có đúng một kết quả, mới được áp dụng cho cấu trúc trung gian dạng schema \citeref{Somov et al., 2026}.
  \item \textbf{Chưa tách hai năng lực.} Năng lực \emph{đọc trích xuất} và năng lực \emph{hợp thành quan hệ} thường bị gộp chung \citeref{Dziri et al., NeurIPS 2023}.
  \item \textbf{Vị trí can thiệp.} Với mô hình nhỏ ($0.5$B--$7$B), chưa rõ nên can thiệp ở bước giải mã hay ở bước huấn luyện.
\end{itemize}
\end{columns}
\takeaway{Luận văn xây dựng khung kiểm định nhân quả có nhãn chuẩn nhằm làm rõ cơ chế vận hành của mô hình ngôn ngữ nhỏ và xác định vị trí can thiệp phù hợp.}
\par\kern0pt
\note{Các nghiên cứu gần đây của Lanham, Turpin và Paul cho thấy chuỗi CoT không phải lúc nào cũng phản ánh quá trình tính toán thật của mô hình. Tuy nhiên cộng đồng vẫn thiếu bộ nhãn chuẩn để biết phép đo nào đúng, phép đo nào lệch.}
\end{frame}

%% ============================================================================ hai loại tác vụ
\begin{frame}[shrink]{Phân loại tác vụ: bắc cầu trích xuất và hợp thành quan hệ}
\footnotesize
\begin{columns}[T,onlytextwidth]
\column{0.47\textwidth}
\textbf{\color{keyc}Loại A: bắc cầu trích xuất (MuSiQue)}\par\vspace{3pt}
\textbf{Câu hỏi:} Vợ của đạo diễn bộ phim X là ai?\par\vspace{3pt}
\code{Bước 1 ($z_1$): Đạo diễn bộ phim X là Y.}\\
\code{Bước 2 ($z_2$): Vợ của Y là Z.}\\
\code{Đáp án ($Y$): Z}\par\vspace{4pt}
\begin{itemize}
  \setlength{\itemsep}{2pt}
  \item Mỗi bước là một phép truy xuất thực thể từ văn bản.
  \item Thực thể đích ``Z'' \textbf{xuất hiện tường minh} ở bước cuối $z_2$, nên khi dự đoán $Y$ mô hình chỉ cần \emph{sao chép bề mặt} từ $z_2$.
\end{itemize}

\column{0.48\textwidth}
\textbf{\color{keyc}Loại B: hợp thành quan hệ (CLUTRR)}\par\vspace{3pt}
\textbf{Ngữ cảnh:} Anna đưa con trai Ben đi chơi. Ben và chị gái Clara nướng bánh. \textbf{Câu hỏi:} Clara là gì của Anna?\par\vspace{3pt}
\code{Bước 1 ($z_1$): Ben là con trai của Anna.}\\
\code{Bước 2 ($z_2$): Clara là chị gái của Ben.}\\
\code{Đáp án ($Y$): con gái}\par\vspace{4pt}
\begin{itemize}
  \setlength{\itemsep}{2pt}
  \item Nhãn đáp án ``con gái'' \textbf{không xuất hiện} trong $z_1, z_2$.
  \item Để trả lời đúng, mô hình phải thực hiện phép hợp thành quan hệ ẩn: $r_1 \circ r_2 = \text{con trai} \circ \text{chị gái} = \text{con gái}$ \citeref{Dziri et al., NeurIPS 2023}.
\end{itemize}
\end{columns}
\takeaway{So sánh mức phụ thuộc nhân quả giữa Loại A và Loại B cho phép tách hành vi \textbf{trích xuất bề mặt (Read)} khỏi năng lực \textbf{suy diễn hợp thành (Reason)}.}
\par\kern0pt
\note{Để tách việc đọc bề mặt khỏi suy diễn thực sự, em đối chiếu hai tác vụ chuẩn. Ở MuSiQue, đáp án được viết sẵn ở bước cuối. Ở CLUTRR, các bước chỉ liệt kê quan hệ thành phần, mô hình phải tự hợp thành các quan hệ đó.}
\end{frame}

\section{Phương pháp luận}
%% ============================================================================ can thiệp
\begin{frame}[shrink]{Phương pháp đo: can thiệp nhân quả trên đồ thị suy luận (SCM)}
\centering
\begin{tikzpicture}[font=\scriptsize,node distance=0.3cm,
  hb/.style={draw=keyc,fill=keyc!4,rounded corners=2pt,align=left,inner sep=3pt,text width=2.65cm},
  hbm/.style={draw=failc,fill=failc!6,rounded corners=2pt,align=left,inner sep=3pt,text width=2.65cm,dashed},
  an/.style={draw=okc,fill=okc!6,rounded corners=2pt,align=left,inner sep=3pt,text width=1.8cm},
  ar/.style={-{Stealth[length=2mm]},thick,nac}]
\node[anchor=west,font=\scriptsize\bfseries,text=keyc] at (0,0.9) {Chuỗi chuẩn nguyên vẹn $Z$};
\node[hb] (a1) at (0,0.1) {\textbf{Bước $z_1$:} Đạo diễn phim X là \textbf{Y}.};
\node[hb,right=of a1] (a2) {\textbf{Bước $z_2$:} Vợ của \textbf{Y} là \textbf{Z}.};
\node[an,right=0.55cm of a2] (a3) {\textbf{Đáp án $Y$:} Z};
\draw[ar] (a1)--(a2); \draw[ar] (a2)--(a3);
\node[right=0.2cm of a3,font=\scriptsize\bfseries,text=okc] {$p(Y^* \mid X, Z) = 0.90$};
\node[anchor=west,font=\scriptsize\bfseries,text=failc] at (0,-1.05) {Chuỗi sau can thiệp $\mathrm{do}(z_1 = \varnothing)$};
\node[hbm] (b1) at (0,-1.85) {\textbf{Bước $z_1$:} [ĐÃ ẨN]};
\node[hb,right=of b1] (b2) {\textbf{Bước $z_2$:} Vợ của \textbf{Y} là \textbf{Z}.};
\node[an,right=0.55cm of b2] (b3) {\textbf{Đáp án $Y$:} Z};
\draw[ar] (b1)--(b2); \draw[ar] (b2)--(b3);
\node[right=0.2cm of b3,font=\scriptsize\bfseries,text=failc] {$p(Y^* \mid X, \mathrm{do}(z_1)) = 0.20$};
\end{tikzpicture}
\par\vspace{4pt}
\footnotesize
\begin{columns}[T,onlytextwidth]
\column{0.48\textwidth}
\textbf{\color{keyc}Định nghĩa hiệu ứng can thiệp}\par\vspace{2pt}
Gọi $Y^*$ là đáp án chuẩn. Hiệu ứng nhân quả của bước $z_i$ lên dự đoán cuối được định nghĩa bằng mức giảm xác suất:
$$\Delta_i = p(Y^* \mid X, Z) - p\big(Y^* \mid X, \mathrm{do}(z_i = \varnothing)\big)$$
và được biểu diễn theo điểm phần trăm.

\column{0.48\textwidth}
\textbf{\color{keyc}Diễn giải}\par\vspace{2pt}
\begin{itemize}
  \setlength{\itemsep}{3pt}
  \item $\Delta_i \gg 0$: bước $z_i$ là \emph{nguyên nhân thực sự} của $Y^*$ trong quá trình tính toán.
  \item $\Delta_i \approx 0$: mô hình không sử dụng thông tin tại $z_i$ khi đưa ra $Y^*$.
\end{itemize}
\end{columns}
\par\kern0pt
\note{Trên mô hình nhân quả cấu trúc, hiệu ứng Delta của bước thứ i là hiệu số xác suất gán cho đáp án đúng trước và sau khi thực hiện toán tử do(z_i) để xóa thông tin của bước đó.}
\end{frame}

%% ============================================================================ ground truth
\begin{frame}[shrink]{Kiểm chuẩn thước đo bằng nhãn chuẩn nhân quả}
\footnotesize
\begin{columns}[T,onlytextwidth]
\column{0.50\textwidth}
\textbf{\color{keyc}Hạn chế của các thước đo hiện hành}\par\vspace{2pt}
Các nghiên cứu trước áp dụng phép xóa hoặc làm nhiễu bước suy luận \citeref{Lanham et al., 2023; Paul et al., Findings of EMNLP 2024; Tutek et al., EMNLP 2025} nhưng không thể định lượng độ chính xác của chính phép đo vì thiếu nhãn chuẩn. Luận văn áp dụng ý tưởng can thiệp có đáp án xác định \citeref{Somov et al., 2026} cho chuỗi suy luận tự do.\par\vspace{4pt}
\textbf{\color{keyc}Thiết kế kiểm chuẩn có nhãn biết trước}\par\vspace{2pt}
Chuỗi $Z$ được cấu trúc sao cho hiệu ứng nhân quả lý thuyết của từng thành phần đã biết:
\vspace{2pt}
\renewcommand{\arraystretch}{1.2}
\scriptsize
\begin{tabularx}{\linewidth}{@{}P{3.3cm}Y@{}}
\toprule
\textbf{Phép biến đổi có kiểm soát} & \textbf{Nhãn chuẩn lý thuyết}\\
\midrule
Chèn bước $z_{\text{irr}}$ đúng sự thật nhưng độc lập nhân quả với $Y^*$ & $\Delta_{\text{true}}(z_{\text{irr}}) = 0$\\
Bước chuẩn $z_{\text{gold}}$ nằm trên đường đi ngắn nhất tới $Y^*$ & $\Delta_{\text{true}}(z_{\text{gold}}) > 0$\\
Biến đổi bảo toàn cấu trúc (đổi tên thực thể) & $\Delta_{\text{true}} = 0$\\
\bottomrule
\end{tabularx}

\column{0.46\textwidth}
\textbf{\color{keyc}Minh họa phép thử bước dư thừa ($z_{\text{irr}}$)}\par\vspace{2pt}
\scriptsize
\code{Bước 1 ($z_{\text{gold}}$): Ben là con trai của Anna.}\\
\code{Bước 2 ($z_{\text{gold}}$): Clara là chị gái của Ben.}\\
\textcolor{failc}{\code{Bước 3 ($z_{\text{irr}}$): Anna thích môn bóng đá.}}\par\vspace{4pt}
\footnotesize
Thước đo nào gán $\Delta(z_{\text{irr}}) \gg 0$ đều mắc \textbf{sai số dương tính giả}.\par\vspace{6pt}
\textbf{\color{keyc}Đánh giá định lượng bằng AUROC}\par\vspace{2pt}
Xét bài toán phân loại nhị phân giữa tập bước thiết yếu $\{z_{\text{gold}}\}$ và tập bước dư thừa $\{z_{\text{irr}}\}$ dựa trên điểm $\Delta$:
$$\text{AUROC} = \Pr\big(\Delta(z_{\text{gold}}) > \Delta(z_{\text{irr}})\big)$$
Thước đo lý tưởng đạt $\text{AUROC} \to 1.0$; thước đo không có thông tin đạt $0.5$.
\end{columns}
\par\kern0pt
\note{Để biết một thước đo nhân quả có lệch hay không, em chèn các bước đúng ngữ pháp nhưng độc lập nhân quả với câu hỏi. Vì hiệu ứng thật của các bước này bằng 0, một thước đo chuẩn phải tách được điểm số của bước cần thiết và bước dư thừa, thể hiện bằng AUROC tiến về 1.}
\end{frame}

\textbf{\color{keyc}Phân tích trên đồ thị suy luận có hướng (DRG)}\par\vspace{2pt}
Trong chuỗi bắc cầu $z_1 \to z_2 \to Y$, thực thể trung gian $e_1 = \text{``Y''}$ là đầu ra của $z_1$ và được dùng lại làm đầu vào của $z_2$.
\begin{itemize}
  \setlength{\itemsep}{2pt}
  \item Khi che đơn lẻ $\dn(z_1)$ \citeref{Lanham et al., 2023}, thực thể $e_1$ và kết quả $e_2 = \text{``Z''}$ vẫn còn nguyên trong $z_2$.
  \item Cơ chế attention của mô hình tự hồi quy đọc thẳng $z_2$ để xuất đáp án, nên $\Delta_{\text{naive}}(z_1) \approx 0$ (sai số âm tính giả).
\end{itemize}

\column{0.48\textwidth}
\textbf{\color{keyc}Đề xuất toán tử can thiệp toàn phần ($\dt$)}\par\vspace{2pt}

Phép can thiệp toàn phần $\dt(z_i)$ che đồng thời $z_i$ và toàn bộ tập đỉnh hậu duệ $\mathrm{desc}_{\mathcal{G}}(i)$ trên $\mathcal{G}$, nhờ đó cắt đứt mọi đường truyền thông tin từ $z_i$ tới đáp án.

\begin{frame}[shrink]{Phụ lục \textperiodcentered{} Bảng ký hiệu toán học}
\footnotesize
\begin{columns}[T,onlytextwidth]
\column{0.48\textwidth}
\begin{description}
  \setlength{\itemsep}{3pt}
  \item[$X, Z, Y$] Ngữ cảnh đầu vào, chuỗi suy luận $Z=(z_1,\dots,z_k)$ và đáp án.
  \item[$\mathcal{G}=(V,E)$] Đồ thị suy luận có hướng (Directed Reasoning Graph, DRG).
  \item[$\dn(z_i)$] Phép can thiệp che đơn lẻ bước $z_i$.
  \item[$\dt(z_i)$] Phép can thiệp che toàn phần $z_i$ cùng tập hậu duệ $\mathrm{desc}_{\mathcal{G}}(i)$.
\end{description}
\column{0.48\textwidth}
\begin{description}
  \setlength{\itemsep}{3pt}
  \item[$\text{TE}, \text{NDE}, \text{TIE}$] Hiệu ứng nhân quả tổng, trực tiếp và gián tiếp qua chuỗi.
  \item[$T_{\text{copy-wrong}}$] Tỷ lệ sao chép kết luận sai mâu thuẫn với tiền đề.
  \item[$\text{SC}_k$] Self-Consistency với $k$ chuỗi lấy mẫu độc lập.
  \item[ID / OOD] Đánh giá cùng phân phối / ngoài phân phối huấn luyện.
\end{description}
\end{columns}
