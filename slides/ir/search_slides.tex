\begin{frame}{Crawl: nguồn URL ban đầu và cách mở rộng}
\small
\muctieu{Bắt đầu từ một tập URL mầm, sau đó tự tìm thêm trang mới qua liên kết.}
\begin{enumerate}
  \setlength{\itemsep}{4pt}
  \item \textbf{URL mầm.} File \code{link\_all.txt} gồm khoảng 100 nghìn URL thuộc hust.edu.vn (trang chủ các đơn vị và trang con). Script \code{feed.py} gửi theo lô (mặc định 5000 URL) vào mediator qua \code{POST /add}. Nạp lại nhiều lần không gây trùng.
  \item \textbf{Mở rộng theo liên kết.} Với mỗi trang đã tải, reader lấy mọi thẻ \code{<a href>}, giữ các URL thuộc hust.edu.vn và gửi lại mediator. URL chưa có trong hàng đợi được thêm vào và tải ở các lượt sau.
  \item \textbf{Phạm vi.} Chỉ thu thập miền hust.edu.vn; liên kết ra ngoài được ghi vào đồ thị nhưng không tải.
\end{enumerate}
\vspace{4pt}
\begin{itemize}
  \item Hệ thống không đọc \code{sitemap.xml}: phạm vi crawl do danh sách mầm và đồ thị liên kết quyết định.
  \item Ví dụ: tải \ex{https://hust.edu.vn/} cho ra các liên kết tới trang tin tức, trang đơn vị; các liên kết này vào hàng đợi rồi lần lượt được tải.
\end{itemize}
\end{frame}

%% ============================================================================ 4
\begin{frame}[shrink]{Crawl: hàng đợi URL và lập lịch}
\small
\muctieu{Quyết định URL nào được tải và lúc nào, bảo đảm không tải trùng và không gửi dồn dập vào một website.}
\begin{columns}[T,onlytextwidth]
\column{0.5\textwidth}
\textbf{1. Chuẩn hoá URL}
\begin{itemize}
  \item Chữ thường scheme và host, bỏ cổng mặc định, bỏ \ex{..}, bỏ \ex{\#...}, bỏ tham số theo dõi (\ex{utm\_*}, \ex{fbclid}), sắp xếp tham số. Ví dụ ở cuối slide.
\end{itemize}
\textbf{2. Hàng đợi trên SQLite}
\begin{itemize}
  \item Bảng \code{urls}: \code{url}, \code{host}, \code{kind}, \code{next\_due\_at}, \code{attempts}. Bảng \code{hosts}: thời điểm gửi gần nhất của mỗi host.
  \item URL đã có thì giữ nguyên lịch cũ, nên gửi lại không gây tải trùng. Hàng đợi nằm trên đĩa, tắt máy không mất.
\end{itemize}
\column{0.46\textwidth}
\textbf{3. Chọn URL để giao}
\begin{itemize}
  \item Điều kiện: đã đến hạn (\code{next\_due\_at} $\le$ hiện tại) và host đã nghỉ đủ 2 giây kể từ lần gửi trước (crawl lịch sự).
  \item Thứ tự: tài liệu trước, sau đó URL có hạn sớm nhất.
\end{itemize}
\textbf{4. Giao cho downloader}
\begin{itemize}
  \item 4 bản downloader, chia lần lượt theo vòng.
  \item Hỏi \code{GET /ping}: rảnh mới gửi \code{POST /load}. Mỗi downloader tải một URL tại một thời điểm.
  \item Không trả lời trong 2 giây hoặc lỗi: tạm ngừng hỏi downloader đó 30 giây.
\end{itemize}
\end{columns}
\vspace{4pt}
\textbf{Ví dụ chuẩn hoá:} \ex{HTTPS://Hust.edu.vn:443/a/../b?z=2\&a=1\&utm\_source=fb\#top} $\rightarrow$ \ex{https://hust.edu.vn/b?a=1\&z=2}
\end{frame}

%% ============================================================================ 5
\begin{frame}[shrink]{Crawl: tải trang và bóc nội dung}
\small
\begin{columns}[T,onlytextwidth]
\column{0.46\textwidth}
\textbf{Downloader}
\begin{itemize}
  \item Trang web: trình duyệt Chromium headless (Puppeteer), chờ sự kiện \code{load} rồi chờ mạng yên tối đa 2 giây để nội dung tải bằng JavaScript hiện ra.
  \item Tài liệu và ảnh (pdf, docx, jpg...): tải trực tiếp bằng \code{fetch}, mã hoá base64.
  \item Giới hạn 30 giây mỗi trang, 50 MB mỗi tệp. Lỗi chỉ ghi log, không gửi tiếp.
\end{itemize}
\textbf{Reader với tài liệu}
\begin{itemize}
  \item Apache Tika bóc chữ của pdf, docx, xlsx. Ảnh không OCR, chỉ lấy tiêu đề và metadata.
\end{itemize}
\column{0.5\textwidth}
\textbf{Reader với trang web: tìm khối nội dung chính}
\begin{itemize}
  \item Bỏ \code{script}, \code{style}, \code{iframe}, phần tử ẩn.
  \item Mỗi đoạn chữ có ít nhất 25 ký tự (không tính thẻ \code{a}) cộng số ký tự của nó cho khối cha, và một nửa cho khối ông.
  \item Khối có tổng điểm cao nhất là nội dung chính; menu, chân trang có ít chữ liền mạch nên bị loại.
\end{itemize}
\textbf{Tiêu đề và liên kết}
\begin{itemize}
  \item Tiêu đề: \code{og:title}, rồi \code{h1}, rồi thẻ \code{title}.
  \item Liên kết ra lấy từ toàn trang; văn bản liên kết là câu chứa thẻ \code{a}.
\end{itemize}
\end{columns}
\vspace{4pt}
\begin{itemize}
  \item Storage ghi bản ghi vào \code{html}, \code{other} hoặc \code{picture}, và mỗi liên kết thành một cạnh trong \code{graph}.
\end{itemize}
\end{frame}

\begin{frame}[shrink]{Crawl: cập nhật trang mới và trang đã thay đổi}
\small
\muctieu{Phát hiện trang mới đăng và cập nhật nội dung đã thay đổi mà không phải crawl lại toàn bộ.}
\textbf{1. Kiểm tra lại định kỳ theo loại URL}\par\vspace{2pt}
\footnotesize
\renewcommand{\arraystretch}{1.25}
\begin{tabularx}{\textwidth}{@{}P{2.4cm}YP{3.0cm}@{}}
\toprule
\textbf{Loại} & \textbf{Nhận biết} & \textbf{Tải lại sau}\\
\midrule
Trang chủ (hub) & Đường dẫn rỗng hoặc \ex{/} & 1 ngày\\
Trang thường & Còn lại & 7 ngày\\
Tài liệu, ảnh & Đuôi \ex{.pdf}, \ex{.docx}, \ex{.jpg}, ... & 30 ngày\\
\bottomrule
\end{tabularx}
\small
\vspace{4pt}
\begin{itemize}
  \item \textbf{Trang mới:} trang chủ được tải lại mỗi ngày. Bài mới đăng xuất hiện dưới dạng liên kết trên trang chủ, reader gửi URL đó về mediator và nó được tải ngay ở lượt sau.
  \item \textbf{Trang thay đổi:} khi tải lại, storage ghi đè nội dung và cập nhật \code{updatedAt}, giữ nguyên \code{createdAt}. Cạnh liên kết mới được thêm vào \code{graph}.
  \item \textbf{URL lỗi:} thử lại ngay ở lượt sau; lỗi 5 lần liên tiếp thì lùi một chu kỳ.
  \item \textbf{Chỉ mục:} search không cập nhật từng phần; gọi \code{POST /reindex} để dựng lại chỉ mục từ dữ liệu mới.
  \item Hệ thống không so sánh nội dung trước khi tải (ETag, Last-Modified): mỗi lần kiểm tra lại đều tải toàn bộ trang.
\end{itemize}
\end{frame}

\begin{frame}{Search: phân tích văn bản tiếng Việt}
\small
\muctieu{Khớp được từ ghép tiếng Việt và truy vấn gõ không dấu mà không cần thư viện tách từ.}
\begin{itemize}
  \item Chuẩn hoá Unicode NFC cho mọi văn bản, vì cùng một chữ có dấu có hai cách mã hoá.
  \item Mỗi âm tiết là một token. Từ ghép được biểu diễn bằng bigram (cặp hai âm tiết liền nhau).
  \item Mỗi trường được lưu thêm một bản bỏ dấu.
\end{itemize}
\vspace{2pt}
\footnotesize
\renewcommand{\arraystretch}{1.3}
\begin{tabularx}{\textwidth}{@{}P{1.8cm}P{4.4cm}Y@{}}
\toprule
\textbf{Analyzer} & \textbf{Xử lý} & \textbf{Ví dụ: Bách Khoa Hà Nội}\\
\midrule
\code{vi} & tách âm tiết, chữ thường & \ex{bách}, \ex{khoa}, \ex{hà}, \ex{nội}\\
\code{viBi} & ghép bigram & \ex{bách khoa}, \ex{khoa hà}, \ex{hà nội}\\
\code{viAscii} & bỏ dấu, đ thành d & \ex{bach}, \ex{khoa}, \ex{ha}, \ex{noi}\\
\code{urlTokens} & tách url theo ký tự không phải chữ, số & \ex{.../hoc-phi-654580.html} thành \ex{hoc}, \ex{phi}, \ex{654580}, \ex{html}\\
\bottomrule
\end{tabularx}
\small
\vspace{4pt}
\begin{itemize}
  \item Truy vấn học phí khớp hai unigram \ex{học}, \ex{phí} và bigram \ex{học phí}; trang chứa đúng cụm học phí được nhiều điểm hơn trang chỉ chứa rời hai âm tiết.
\end{itemize}
\end{frame}

\begin{frame}[shrink]{Search: các trường của chỉ mục và hàm điểm BM25}
\small
\muctieu{Mỗi phần của tài liệu được lưu thành trường riêng để chấm điểm với trọng số khác nhau.}
\footnotesize
\renewcommand{\arraystretch}{1.22}
\begin{tabularx}{\textwidth}{@{}P{3.4cm}P{1.9cm}Y@{}}
\toprule
\textbf{Trường} & \textbf{Analyzer} & \textbf{Dùng để}\\
\midrule
\code{title}, \code{content} & \code{vi} & Khớp từng âm tiết. \code{content} lưu thêm vị trí từ cho truy vấn cụm và đoạn trích\\
\code{title\_bi}, \code{content\_bi} & \code{viBi} & Khớp cụm hai âm tiết\\
\code{title\_ascii}, \code{content\_ascii} & \code{viAscii} & Khớp truy vấn không dấu\\
\code{url\_tokens} & \code{urlTokens} & Khớp từ trong url\\
\code{url}, \code{host}, \code{collection} & nguyên văn & Định danh, lọc theo website và loại tài liệu\\
\code{sig:pagerank}, \code{sig:recency} & số & Tín hiệu ngoài văn bản\\
\bottomrule
\end{tabularx}
\small
\vspace{4pt}
\textbf{BM25}, tính riêng trên từng trường:
$$\mathrm{BM25}(q,d)=\sum_{t\in q}\mathrm{IDF}(t)\,\frac{f(t,d)\,(k_1+1)}{f(t,d)+k_1\big(1-b+b\,\frac{|d|}{\mathrm{avgdl}}\big)}$$
\begin{itemize}
  \item $f(t,d)$: số lần từ $t$ xuất hiện trong trường; $|d|$: độ dài trường; $\mathrm{avgdl}$: độ dài trung bình.
  \item $k_1 = 1.2$. $b = 0.75$ cho tiêu đề và url; $b = 0.5$ cho nội dung, để tài liệu dài như PDF không bị phạt độ dài quá nặng.
\end{itemize}
\end{frame}

\end{itemize}

\end{itemize}
\textbf{Bộ lọc}
\begin{itemize}
  \item Lọc theo loại tài liệu và theo website (ví dụ chỉ soict.hust.edu.vn), không ảnh hưởng đến điểm.
\end{itemize}
\end{columns}
\end{frame}

\begin{frame}[shrink]{Search giai đoạn 2: chấm lại 100 kết quả đầu}
\small
\muctieu{Sắp xếp lại 100 kết quả bằng các tín hiệu chính xác hơn nhưng tốn kém hơn BM25.}
\textbf{Điểm mới} = điểm BM25 + điểm của các tín hiệu sau:\par\vspace{2pt}
\footnotesize
\renewcommand{\arraystretch}{1.3}
\begin{tabularx}{\textwidth}{@{}P{3.6cm}Q{1.5cm}Y@{}}
\toprule
\textbf{Tín hiệu} & \textbf{Trọng số} & \textbf{Ý nghĩa}\\
\midrule
Cụm từ trong tiêu đề & 2 & Các từ của truy vấn đứng gần nhau đúng thứ tự, cho phép chèn tối đa 2 từ\\
Cụm từ trong nội dung & 1 & Như trên\\
PageRank & 2 & Trang được nhiều trang khác trỏ tới\\
Độ mới (khi \code{fresh=1}) & 2 & Trang mới cập nhật\\
\bottomrule
\end{tabularx}
\small
\vspace{4pt}
\textbf{Chuẩn hoá tín hiệu bằng hàm bão hoà}
$$w\cdot\frac{s}{s+\mathrm{pivot}}, \qquad s_{\text{PageRank}}=\max(\mathrm{PR}\times N,\,10^{-4}), \qquad s_{\text{độ mới}}=\frac{1}{1+\text{số ngày từ lần cập nhật}/30}$$
\begin{itemize}
  \item Hàm tăng nhanh rồi phẳng dần: một trang có PageRank rất lớn vẫn không lấn át được điểm văn bản.
  \item $\mathrm{PR}\times N$ đưa giá trị trung bình về 1, không phụ thuộc số trang. Độ mới bằng 0.5 sau 30 ngày.
\end{itemize}
\end{frame}

%% ============================================================================ 12
\begin{frame}{Search: PageRank trên đồ thị liên kết}
\small
\muctieu{Đo mức quan trọng của trang dựa trên cấu trúc liên kết: trang được nhiều trang quan trọng trỏ tới thì quan trọng.}
\textbf{Mô hình.} Người duyệt web ngẫu nhiên: với xác suất 0.85 bấm một liên kết trên trang hiện tại, với xác suất 0.15 nhảy tới một trang bất kỳ.
$$\mathrm{PR}(v)=\frac{0.15}{N}+0.85\left(\sum_{u\to v}\frac{\mathrm{PR}(u)}{\mathrm{outdeg}(u)}+\frac{\mathrm{dangling}}{N}\right)$$
\begin{itemize}
  \item $\mathrm{dangling}$: tổng PageRank của các trang không có liên kết ra, chia đều cho mọi trang để tổng luôn bằng 1.
  \item Dựng đồ thị từ \code{graph}: mỗi url là một nút, bỏ cạnh tự trỏ và cạnh trùng.
  \item Lặp từ giá trị khởi tạo $1/N$, tối đa 50 vòng, dừng khi tổng thay đổi giữa hai vòng nhỏ hơn $10^{-6}$. Chi phí $O(\text{số vòng}\times(N+E))$.
  \item PageRank dùng ở hai chỗ: tín hiệu chấm lại, và chọn bản giữ lại trong nhóm trang trùng.
\end{itemize}
\end{frame}

%% ============================================================================ 13
\begin{frame}[shrink]{Search: phát hiện trang rác bằng độ lạ của từ vựng}
\small
\muctieu{Loại các trang bị chèn nội dung rác (từ khoá quảng cáo) khỏi kết quả.}
\textbf{Nhận xét.} Trang thật của hust dùng từ vựng chung (đào tạo, sinh viên, thông báo). Trang bị chèn rác dùng nhiều từ hầu như không xuất hiện ở trang khác.\par
\vspace{4pt}
\textbf{Độ lạ của trang} $p$ là IDF trung bình của các từ trong trang:
$$\text{độ lạ}(p)=\frac{1}{|W(p)|}\sum_{w\in W(p)}\log\frac{N}{\mathrm{df}(w)}$$
\begin{itemize}
  \item $N$: số trang html có ít nhất 50 từ; $\mathrm{df}(w)$: số trang chứa từ $w$.
  \item Từ phổ biến cho giá trị gần 0; từ chỉ có ở một trang cho giá trị lớn nhất $\log N$.
\end{itemize}
\textbf{Chọn ngưỡng tự động}
\begin{enumerate}
  \item Lập histogram độ lạ (bin rộng 0.1). Trang thật tạo một cụm lớn, trang rác tạo một cụm nhỏ ở phía cao hơn.
  \item Làm mượt histogram đến khi còn đúng hai đỉnh; ngưỡng là đáy thung lũng giữa hai đỉnh, đo trên histogram gốc (phương pháp Prewitt và Mendelsohn).
  \item Không tìm được hai đỉnh rõ ràng thì không đánh dấu trang nào.
\end{enumerate}
\begin{itemize}
  \item Trang vượt ngưỡng được gắn nhãn \code{spam}. Không cần đặt ngưỡng tay, vì ngưỡng thay đổi theo dữ liệu.
\end{itemize}
\end{frame}

%% ============================================================================ 14
\begin{frame}{Search: khử trang trùng nội dung}
\small
\muctieu{Tránh kết quả lặp do bản sao, bản in hoặc trang mirror có nội dung giống hệt.}
\begin{itemize}
  \item \textbf{Khoá nội dung:} SHA-1 của nội dung sau khi chuẩn hoá NFC, chữ thường, gộp khoảng trắng. Hai trang cùng khoá được coi là trùng.
  \item Nội dung dưới 200 ký tự không tính khoá, vì quá ngắn để kết luận là trùng.
\end{itemize}
\vspace{4pt}
\textbf{Chọn một bản giữ lại trong mỗi nhóm trùng}
\begin{enumerate}
  \item Trang có PageRank cao nhất.
  \item Hoà thì chọn url ngắn hơn.
  \item Trang đã bị đánh dấu rác không được chọn.
\end{enumerate}
\vspace{4pt}
\begin{itemize}
  \item Các trang còn lại trong nhóm được gắn nhãn \code{dup} và bị loại khi truy vấn.
  \item Ví dụ: cùng một thông báo được đăng ở trang tin tức và trang đơn vị; chỉ trang có PageRank cao hơn xuất hiện trong kết quả.
\end{itemize}
\end{frame}

\begin{frame}{Search: bốn cấu hình xếp hạng}
\small
\muctieu{Bật lần lượt từng kỹ thuật để đo đóng góp của mỗi kỹ thuật trên cùng một chỉ mục.}
\renewcommand{\arraystretch}{1.4}
\begin{tabularx}{\textwidth}{@{}P{1.5cm}YP{3.3cm}P{2.3cm}@{}}
\toprule
\textbf{Profile} & \textbf{Truy hồi} & \textbf{Chấm lại} & \textbf{Loại rác, trùng}\\
\midrule
A & BM25 trên tiêu đề, nội dung, url & không & không\\
B & A + bigram + bỏ dấu & không & không\\
C & như B & cụm từ, PageRank & không\\
D (mặc định) & như B & như C & có\\
\bottomrule
\end{tabularx}
\vspace{6pt}
\begin{itemize}
  \item Ví dụ truy vấn học phí với profile D: truy hồi trên 7 trường, loại trang rác và trùng, lấy 100 kết quả; trang có cụm học phí trong tiêu đề và trang có PageRank cao được cộng điểm; trả 10 kết quả đầu kèm đoạn trích có từ khoá được tô đậm.
\end{itemize}
\end{frame}

%% ============================================================================ 16
\begin{frame}[shrink]{Search: phương pháp đánh giá}
\small
\muctieu{So sánh các profile A đến D bằng nhãn liên quan do người gán.}
\begin{itemize}
  \item \textbf{Truy vấn:} 25 truy vấn gồm có dấu, không dấu và tên đơn vị, ví dụ \ex{học phí 2025}, \ex{lich thi}, \ex{soict tuyển sinh}.
  \item \textbf{Nhãn:} 0 không liên quan, 1 liên quan một phần, 2 liên quan hoàn toàn.
  \item \textbf{Pooling:} gộp 10 kết quả đầu của cả 4 profile để gán nhãn, nên mọi trang được bất kỳ profile nào xếp cao đều được xem xét.
\end{itemize}
\vspace{4pt}
\textbf{Độ đo trên 10 kết quả đầu}
$$\mathrm{P@10}=\frac{\text{số kết quả có nhãn}\ge 1}{10}, \qquad \mathrm{nDCG@10}=\frac{1}{\mathrm{IDCG@10}}\sum_{i=1}^{10}\frac{2^{g_i}-1}{\log_2(i+1)}$$
\begin{itemize}
  \item P@10 đo tỷ lệ kết quả liên quan. nDCG@10 thưởng nhiều hơn khi trang liên quan hoàn toàn đứng ở vị trí cao.
  \item So sánh B với A cho thấy đóng góp của bigram và bỏ dấu; C với B: cụm từ và PageRank; D với C: loại rác và trùng.
\end{itemize}
\end{frame}

\begin{frame}{Giới hạn và hướng mở rộng}
\begin{itemize}
  \setlength{\itemsep}{5pt}
  \item \textbf{Crawl:} chưa đọc \code{sitemap.xml}; chưa dùng ETag hoặc Last-Modified nên mỗi lần kiểm tra lại đều tải toàn bộ trang.
  \item \textbf{Chỉ mục:} dựng lại toàn bộ mỗi lần, chưa cập nhật từng phần; tín hiệu độ mới lệch dần nếu không dựng lại.
  \item \textbf{Tách từ:} dùng bigram thay cho tách từ tiếng Việt; hướng mở rộng là dùng pyvi.
  \item \textbf{PageRank:} mọi liên kết có trọng số như nhau, nên liên kết ở menu và chân trang làm tăng điểm trang chủ; chưa dùng anchor text.
  \item \textbf{Trang rác:} không phát hiện được nếu số trang rác quá ít để tạo cụm riêng trên histogram.
  \item \textbf{Tính năng:} chưa có gợi ý khi gõ và sửa chính tả.
\end{itemize}
\end{frame}
